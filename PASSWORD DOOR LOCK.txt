
//Simple password doorlock 
//Pin config keypad pin 1 - Digital pin 9, 2-8, 3-7, 4-6, 5-5, 6-4, 7-3, 8-2\
//Pin config Servo - pin 10, Irsensor - Pin 11
            

#include <Keypad.h>
#include <Servo.h>

// Keypad layout
const byte ROWS = 4; 
const byte COLS = 4; 
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

// Connect to the row and column pinouts of the keypad
byte rowPins[ROWS] = {9, 8, 7, 6}; 
byte colPins[COLS] = {5, 4, 3, 2}; 

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

Servo lockServo;
const int servoPin = 10;
const int irSensorPin = 11;

String secretCode = "1234"; // Change your password here
String inputPassword = "";

bool isUnlocked = false;

void setup() {
  Serial.begin(9600);
  
  lockServo.attach(servoPin);
  lockServo.write(0); // 0 degrees,Locked position
  
  pinMode(irSensorPin, INPUT);
  
  printWelcomeMessage();
}

void loop() {
  char key = keypad.getKey();

  if (key) {
    if (key == '#') {
      // Evaluate password when '#' is pressed
      checkPassword();
    } 
    else if (key == '*') {
      // Clear input when '*' is pressed
      inputPassword = "";
      Serial.println("\nPassword cleared.");
    } 
    else {
      // Append character to input password (max 4 digits)
      if (inputPassword.length() < 4) {
        inputPassword += key;
        Serial.print("*"); // Print asterisk for security
      }
    }
  }

  // Auto-relock logic using the IR sensor
  // Assuming IR sensor outputs LOW when the door is closed (obstacle detected)
  if (isUnlocked) {
    int irState = digitalRead(irSensorPin);
    
    // If the door is opened and then closed again, lock automatically after 3 seconds
    if (irState == LOW) { 
      delay(10000); // Wait 10 seconds to let someone walk through
      lockServo.write(0); // Lock the door
      isUnlocked = false;
      Serial.println("\nDoor closed and automatically LOCKED.");
      printWelcomeMessage();
    }
  }
}

void checkPassword() {
  if (inputPassword == secretCode) {
    Serial.println("\nAccess Granted! Door Unlocked.");
    lockServo.write(90); // 90 degrees = Unlocked position
    isUnlocked = true;
    inputPassword = "";
  } else {
    Serial.println("\nAccess Denied! Incorrect Password.");
    inputPassword = "";
  }
}

void printWelcomeMessage() {
  Serial.println("============================");
  Serial.println("Enter 4-digit code & press #");
  Serial.println("Press * to clear input");
  Serial.println("============================");
}
